```
    for func in dir(code):
        if not func.startswith('__'):
            self.functions.append(func)
```

**Step 3: Extract Classes**

```
def extract_classes(self, code):
    self.classes = [cls for cls in dir(code) if not cls.startswith('__') and isinstance(getattr(code, cls), type)]

# Initialize class extraction
self.extract_classes(EchoFullThrottleRite)
print("Classes:", self.classes)
```

**Step 4: Parse Imports**

```
def parse_imports(self, code):
    self.imports = [imp for imp in dir(code) if imp.startswith('import ') or imp.startswith('from ')]
    # Remove any unnecessary imports (e.g., `time` is not imported explicitly)
    self.imports = list(set([imp.split()[1] for imp in self.imports]))

# Initialize import parsing
self.parse_imports(EchoFullThrottleRite)
print("Imports:", self.imports)
```

**Step 5: Output the Results**

```
def output_results(self):
    print(f"Functions: {', '.join(self.functions)}")
    print(f"Classes: {', '.join(self.classes)}")
    print(f"Imports: {', '.join(self.imports)}")

# Run the analysis
self.output_results()
```
