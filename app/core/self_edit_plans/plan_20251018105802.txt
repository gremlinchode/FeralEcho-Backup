    function_pattern = r"def\s+([a-zA-Z_][a-zA-Z0-9_]*)\((.*?)\):"
    functions = re.findall(function_pattern, parsed_code)
    return [func for func in set(functions) if func]

def extract_imports(parsed_code):
    import_pattern = r"import\s+(.*?)(\,|$)"
    imports = re.findall(import_pattern, parsed_code)
    return [imp.strip() for imp in set(imports) if imp]
