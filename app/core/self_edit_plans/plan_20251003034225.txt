import os
import stat

def check_permissions(file_path):
    # Get file stats using os.stat()
    file_stats = os.stat(file_path)
    
    # Check permissions (e.g., owner, group, others)
    if file_stats.st_mode & stat.S_IRUSR:
        print(f"File '{file_path}' is readable by the owner.")
    else:
        print(f"File '{file_path}' is not readable by the owner.")

# Example usage
