Putting it all together:
```python
def apply_self_edits(text):
    import re

    sentences = re.split(r'[.!?]+', text)

    self_edit_rules = {
        'sentence 1': ['Corrected this', 'Improved that'],
        'sentence 2': ['Removed unnecessary words']
    }

    edited_sentences = []
    for sentence in sentences:
        if sentence in self_edit_rules:
            edits = self_edit_rules[sentence]
            edited_sentence = f"{sentence} {edits[0]} {edits[1]}"
            edited_sentences.append(edited_sentence)
        else:
            edited_sentences.append(sentence)

    final_text = ' '.join(edited_sentences)
    print(final_text)

# Test the function
```
