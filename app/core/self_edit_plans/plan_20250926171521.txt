```python
def apply_self_edits(text_input, self_edit_dict):
    words = text_input.split()
    edited_words = []
    for word in words:
        if word in self_edit_dict:
            edited_word = self_edit_dict[word]
        else:
            edited_word = word
        edited_words.append(edited_word)
    edited_text = ' '.join(edited_words)
    return edited_text

# Example usage:
```
