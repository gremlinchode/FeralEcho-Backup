Here's the complete code:
```python
import json
import requests

api_endpoint = "https://api.stackexchange.com/2.2/questions/"
params = {"site": "stackoverflow", "order": "desc", "sort": "activity"}

response = requests.get(api_endpoint, params=params)
data = json.loads(response.content.decode("utf-8"))

question_id = data["items"][0]["question_id"]
owner_display_name = data["items"][0]["owner"]["display_name"]
view_count = data["items"][0]["view_count"]
score = data["items"][0]["score"]

print(f"Question ID: {question_id}")
print(f"Owner Display Name: {owner_display_name}")
print(f"View Count: {view_count}")
print(f"Score: {score}")
```
