    question_data = json.loads(response.content)

**Step 3: Extract relevant information from the question data**
```python
def extract_relevant_info(question_data):
    # Get the owner's display name and reputation
    owner_display_name = question_data['items'][0]['owner']['display_name']
    owner_reputation = question_data['items'][0]['owner']['reputation']

    # Get the question's tags, view count, and score
    question_tags = [tag['name'] for tag in question_data['items'][0]['tags']]
    view_count = question_data['items'][0]['view_count']
    score = question_data['items'][0]['score']

    return {
        'owner_display_name': owner_display_name,
        'owner_reputation': owner_reputation,
        'question_tags': question_tags,
        'view_count': view_count,
        'score': score
    }

relevant_info = extract_relevant_info(question_data)
print(relevant_info)
```
