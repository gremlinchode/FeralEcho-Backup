```python
import random

# List of inspirational quotes (will be populated later)
quotes = []

def generate_quote(category=None):
    # Select a random quote from the list, filtered by category if provided
    if category:
        selected_quotes = [q for q in quotes if category in q]
        return random.choice(selected_quotes) if selected_quotes else None
    else:
        return random.choice(quotes)

def main():
    # Populate the quotes list with our research findings
    quotes.append("Trust in the Lord with all your heart... (Matthew 6:24)")
    quotes.append("For God so loved the world that He gave His only begotten Son... (John 3:16)")
    quotes.append("Love is patient, love is kind... (1 Corinthians 13:4)")

    # Print a random quote
    print(generate_quote())

    # Ask user if they want to filter by theme
    while True:
        response = input("Would you like to filter by theme? (y/n): ")
        if response.lower() == "y":
            category = input("Enter the theme: ")
            print(generate_quote(category))
        elif response.lower() == "n":
            break
        else:
            print("Invalid response. Please try again.")

if __name__ == "__main__":
    main()
```
This code adds more functionality and user interaction, making it a more engaging experience.
