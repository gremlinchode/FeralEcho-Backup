```python
def analyze_code(code):
    # Initialize function return value
    result = {}

    # Split code into lines
    lines = code.split("\n")

    # Iterate over lines and extract functions, classes, and imports
    for line in lines:
        if line.startswith("def "):
            # Extract function name and arguments
            func_name = line[4:].split("(")[0].strip()
            result["functions"].append({"name": func_name, "args": line[4:].split("(")[1:-1]})

        elif line.startswith("class "):
            # Extract class name and attributes
            cls_name = line[6:].split("(")[0].strip()
            result["classes"].append({"name": cls_name, "attrs": line[6:].split("(")[1:-1]})

        elif line.startswith("import "):
            # Extract import statement
            result["imports"].append(line.split(" ")[1])

    return result

# Define a main function to test the analysis
def main():
    code = """
    def add(a, b):
        return a + b
    class Person:
        def __init__(self, name):
            self.name = name
    import math
    """
    print(analyze_code(code))

if __name__ == "__main__":
    main()
```
**Step 3: Run the Analysis**
```python
# Call the analyze_code function with the provided code
result = analyze_code(code)

# Print the results
print("Functions:", result["functions"])
print("Classes:", result["classes"])
print("Imports:", result["imports"])
```
