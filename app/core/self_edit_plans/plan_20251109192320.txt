import time

# Define a function to schedule tasks
def schedule_task(task_name, deadline):
    # Convert deadline to datetime object
    deadline_datetime = datetime.datetime.strptime(deadline, "%Y-%m-%d %H:%M")
    
    # Calculate the time difference between now and deadline
    time_diff = deadline_datetime - datetime.datetime.now()
    
    # Print a message indicating task is scheduled for that time
    print(f"Task '{task_name}' is scheduled for {deadline} ({time_diff})")

# Test the function
