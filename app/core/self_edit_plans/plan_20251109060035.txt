    # Extract functions, classes, and imports from parsed code
    # Use a dictionary to store the results
    analysis = {
        'functions': [],
        'classes': [],
        'imports': []
    }

    for item in code:
        if isinstance(item, dict) and 'name' in item:
            if item['name'] == 'function':
                analysis['functions'].append(item)
            elif item['name'] == 'class':
                analysis['classes'].append(item)
        elif isinstance(item, list):
            # Recursively parse nested lists
            analysis.update(parse_parsed_code(item))

def extract_function_info(functions):
    for function in functions:
        # Extract function name, parameters, and return type
        function_name = function['name']
        parameters = []
        return_type = None

        if 'parameters' in function:
            parameters = function['parameters']

        if 'return_type' in function:
            return_type = function['return_type']

        yield {
            'name': function_name,
            'parameters': parameters,
            'return_type': return_type
        }

def extract_class_info(classes):
    for class_ in classes:
        # Extract class name, attributes, and methods
        class_name = class_['name']
        attributes = []
        methods = []

        if 'attributes' in class_:
            attributes = class_['attributes']

        if 'methods' in class_:
            methods = class_['methods']

        yield {
            'name': class_name,
            'attributes': attributes,
            'methods': methods
        }
