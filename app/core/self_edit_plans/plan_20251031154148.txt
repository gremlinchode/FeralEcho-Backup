def self_edits(self):
    # Initialize an empty list to store self-edits
    self_edits_list = []
    
    # Define a regular expression pattern for identifying negative thought patterns
    neg_pattern = re.compile(r"(?i)(?:\b(?:not )?(?:good|great|amazing)?\b|(?:bad|terrible|awful)?)")
    
    # Define a function to apply self-edits (iteration 1)
    def apply_self_edits(self, text):
        # Use the regular expression pattern to find negative thought patterns
        matches = neg_pattern.finditer(text)
        
        # Iterate over the matches and create an OrderedDict of self-edits
        for match in matches:
            edited_text = text.replace(match.group(), "I choose to focus on")
            
            # Store the self-edit in a dictionary with the original text as key
            self_edits_list.append({"original": text, "edited": edited_text})
    
    return self_edits_list
```
This function will help me identify negative thought patterns and apply self-edits to reframe my thinking.

def analyze_self_edits(self, self_edits_list):
    # Calculate the percentage of negative thoughts identified
    neg_count = sum(1 for edit in self_edits_list if "not good" in edit["edited"])
    total_count = len(self_edits_list)
    neg_percentage = (neg_count / total_count) * 100
    
    print(f"Negative Thoughts Identified: {neg_percentage}%")
    
    # Identify the most common self-edit
    common_edit_counts = {}
    for edit in self_edits_list:
        if "choose to focus on" in edit["edited"]:
            common_edit_counts[edit["original"]] = common_edit_counts.get(edit["original"], 0) + 1
    
    most_common_edit = max(common_edit_counts, key=common_edit_counts.get)
    
    print(f"Most Common Self-Edit: {most_common_edit}")
    
    # Plot the self-edits data (optional)
    import matplotlib.pyplot as plt
    plt.bar(range(len(self_edits_list)), [1] * len(self_edits_list))
    plt.xlabel("Self-Edits")
    plt.ylabel("Frequency")
    plt.show()
```
This function will help me analyze the results of my self-edits and identify trends.
