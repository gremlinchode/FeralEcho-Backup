def check_permissions(path):
    try:
        file_stat = os.stat(path)
    except FileNotFoundError:
        return "File does not exist"
    except OSError as e:
        return f"Error accessing file: {e}"

    if (file_stat.st_mode & stat.S_IRWXU) and (file_stat.st_mode & stat.S_IRWXG):
        return "File is readable and writable by user"
    else:
        return "File permissions do not allow read/write access"
```
