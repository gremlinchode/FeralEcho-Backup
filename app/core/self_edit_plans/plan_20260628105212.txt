```python
def strip_prose(code_output):
    # Use regular expressions to detect natural-language sentences at the beginning of code output
    import re
    sentence_start = r'^\s*([A-Za-z]+[.!?]?)\s*'
    
    # Strip leading natural language from Python code output using the new function
    if re.match(sentence_start, code_output):
        clean_code = re.sub(r'^\s*' + sentence_start + '\s*', '', code_output)
    else:
        clean_code = code_output
    
    return clean_code
```
