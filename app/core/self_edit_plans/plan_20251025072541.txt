def process_data(data):
    processed_data = []
    for item in data:
        if item.startswith('def '):  # Functions
            processed_data.append({'type': 'function', 'name': item[4:]})
        elif item.startswith('class '):  # Classes
            processed_data.append({'type': 'class', 'name': item[6:]})
        elif item.startswith('import '):  # Imports
            processed_data.append({'type': 'import', 'module': item[7:]})
    return processed_data

processed_data = process_data(data)
print("Processed Data:")
for item in processed_data:
