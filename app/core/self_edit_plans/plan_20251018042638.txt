Example:
```python
# Assume 'parsed_code' is a string containing the parsed code
functions = []
classes = []
imports = []

for line in parsed_code.split('\n'):
    if re.match(r'^def\s+([a-zA-Z_][a-zA-Z0-9_]*)\s*\(', line):
        functions.append(re.match(r'^def\s+([a-zA-Z_][a-zA-Z0-9_]*)\s*\(', line).group(1))
    elif re.match(r'^class\s+([a-zA-Z_][a-zA-Z0-9_]*)\s*:', line):
        classes.append(re.match(r'^class\s+([a-zA-Z_][a-zA-Z0-9_]*)\s*:', line).group(1))
    elif re.match(r'^import\s+([a-zA-Z0-9_.\-]+)$', line):
        imports.append(re.match(r'^import\s+([a-zA-Z0-9_.\-]+)$', line).group(1))

print("Functions:", functions)
print("Classes:", classes)
print("Imports:", imports)
```
