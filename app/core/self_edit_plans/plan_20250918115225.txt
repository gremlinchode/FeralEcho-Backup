```python
def analyze_code(parsed_code):
    functions = []
    classes = []
    imports = []

    for token in parsed_code:
        if token.startswith("def "):  # Function definition
            functions.append(token[4:])  # Extract function name
        elif token.startswith("class "):  # Class definition
            classes.append(token[6:])  # Extract class name
        elif token.startswith("import"):  # Import statement
            imports.append(token[7:])  # Extract import module

    return functions, classes, imports
```
