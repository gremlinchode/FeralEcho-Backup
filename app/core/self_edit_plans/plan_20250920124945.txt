```python
def evaluate_expression(expression):
    # Regular expression pattern to split the input
    pattern = r"(\d+(?:\.\d+)?)([+-*/])"

    # Split the input into operands and operators
    tokens = re.findall(pattern, expression)

    # Initialize variables for calculation
    result = 0
    operator = "+"

    for token in tokens:
        if token[1] == "+":
            result += float(token[0])
        elif token[1] == "-":
            result -= float(token[0])
        elif token[1] == "*":
            result *= float(token[0])
        elif token[1] == "/":
            result /= float(token[0])

    return result
```
