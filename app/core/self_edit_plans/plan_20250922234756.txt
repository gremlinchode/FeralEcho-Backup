def analyze_code(code):
    # Initialize empty lists for functions, classes, and imports
    functions = []
    classes = []
    imports = []

    # Loop through the code lines
    for line in code.split('\n'):
        # Check for function definitions (e.g., def my_function(x):)
        if line.strip().startswith('def '):
            functions.append(line.strip())

        # Check for class definitions (e.g., class MyClass:)
        elif line.strip().startswith('class '):
            classes.append(line.strip())

        # Check for import statements (e.g., import math)
        elif line.strip().startswith('import ') or line.strip().startswith('from '):
            imports.append(line.strip())

    return functions, classes, imports

# Example usage:
code = """
def my_function(x): pass
class MyClass: pass
import math
"""

functions, classes, imports = analyze_code(code)
print("Functions:", functions)
print("Classes:", classes)
print("Imports:", imports)
