```python
def check_permissions(path, access_level, user_id):
    try:
        stat_info = os.stat(path)
        if stat.S_ISDIR(stat_info.st_mode):
            # For directories, we need to check permissions separately
            dir_perm = stat_info.st_mode & 0o666
            return (dir_perm & access_level) == access_level
        else:
            # For files, we can use the `stat` module directly
            file_perm = stat.S_IMODE(stat_info.st_mode)
            return (file_perm & access_level) == access_level
    except FileNotFoundError:
        print(f"File not found: {path}")
        return False  # or raise an exception if you prefer
```
