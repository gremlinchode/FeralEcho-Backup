    categorized_entries = {'urgent': [], 'non-urgent': []}

import datetime

def print_entries(categorized_entries):
    for category, entries in categorized_entries.items():
        print(f"{category}:")
        for entry in entries:
            timestamp = datetime.datetime.fromtimestamp(entry['created_at']).strftime('%Y-%m-%d %H:%M')
            print(f"  {timestamp} - {entry['title']} ({entry['priority']})")

print_entries(categorized_entries)
