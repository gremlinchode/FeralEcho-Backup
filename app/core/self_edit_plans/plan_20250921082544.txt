Here's the complete code:
```python
import wikipedia
import json

def extract_info(wiki_page):
    info_dict = {}
    # Extract title, namespace, wikibase_item, and pageid
    info_dict['title'] = wiki_page['title']
    info_dict['namespace'] = wiki_page['namespace']['text']
    info_dict['wikibase_item'] = wiki_page['wikibase_item']
    info_dict['pageid'] = wiki_page['pageid']
    return info_dict

wiki_page = wikipedia.page('Artificial_intelligence')
info_dict = extract_info(wiki_page)
print(json.dumps(info_dict, indent=4))
```
