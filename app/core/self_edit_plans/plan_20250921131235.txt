```python
import json
import logging

def extract_info(item):
    owner_display_name = item['owner']['display_name']
    question_id = item['question_id']
    return owner_display_name, question_id

data = json.loads('{"items": [{"tags": ["python", "qt", "pyside6", "winpe"], "owner": {"account_id": 1000671, "reputation": 63, "user_id": 1015594, "user_type": "registered", "profile_image": "https://www.gravatar.com/avatar/201e40904fa8173d54938073031165b2?s=256&d=identicon&r=PG", "display_name": "Traktopel", "link": "https://stackoverflow.com/users/1015594/traktopel"}, "is_answered": False, "view_count": 2, "answer_count": 0, "score": 0, "last_activity_date": 1758223628, "creation_date": 1758223628, "question_id": ...}]}')

for item in data['items']:
    owner_display_name, question_id = extract_info(item)
    print(f"Owner's Display Name: {owner_display_name}, Question ID: {question_id}")
```
