```python
def check_permissions(path):
    """
    Check if a file or directory has read, write, and execute permissions.
    
    :param path: The path to the file or directory
    :return: A dictionary with 'read', 'write', and 'execute' as keys indicating permission status (True/False)
    """
    stats = os.stat(path)
    return {
        'read': stat.S_IRUSR & stats.st_mode,
        'write': stat.S_IWUSR & stats.st_mode,
        'execute': stat.S_IXUSR & stats.st_mode
    }
```

```python
def check_permissions(path):
    try:
        stats = os.stat(path)
        return {
            'read': stat.S_IRUSR & stats.st_mode,
            'write': stat.S_IWUSR & stats.st_mode,
            'execute': stat.S_IXUSR & stats.st_mode
        }
    except OSError as e:
        print(f"Error checking permissions for {path}: {e}")
        return {}
```
