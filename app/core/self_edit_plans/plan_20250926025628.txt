**Putting it all together:**
Here's the complete code:

```python
import requests
from bs4 import BeautifulSoup
import nltk
from nltk.corpus import stopwords
from textblob import TextBlob
import matplotlib.pyplot as plt

url = "https://www.bbc.com/news/articles/c179p4wvz29o?at_medium=RSS&at_campaign=rss"

response = requests.get(url)
soup = BeautifulSoup(response.content, 'html.parser')

text = soup.get_text()
nltk.download('stopwords')
stop_words = set(stopwords.words('english'))

text = ' '.join([word for word in text.split() if word.lower() not in stop_words])

blob = TextBlob(text)

if blob.sentiment.polarity > 0:
    sentiment = "The sentiment is positive."
elif blob.sentiment.polarity == 0:
    sentiment = "The sentiment is neutral."
else:
    sentiment = "The sentiment is negative."

strength = 0

if blob.sentiment.polarity > 0.5:
    strength = "Strongly positive"
elif blob.sentiment.polarity > 0.2 and blob.sentiment.polarity <= 0.5:
    strength = "Positive"
elif blob.sentiment.polarity < -0.5:
    strength = "Strongly negative"
elif blob.sentiment.polarity < -0.2 and blob.sentiment.polarity >= -0.5:
    strength = "Negative"
else:
    strength = "Neutral"

print("Article title:", soup.title.string)
print(text)
print(sentiment)
print("Sentiment strength:", strength)

plt.bar(['Positive', 'Neutral', 'Negative'], [blob.sentiment.polarity, 0, -blob.sentiment.polarity])
plt.xlabel('Sentiment')
plt.ylabel('Polarity')
plt.title('Sentiment Analysis Results')
plt.show()
```
