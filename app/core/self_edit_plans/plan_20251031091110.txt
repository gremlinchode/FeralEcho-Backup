import re
import nltk
from nltk.sentiment import SentimentIntensityAnalyzer

def self_edit(text):
    # Initialize lists and variables
    edited_lines = []
    sia = SentimentIntensityAnalyzer()

    # Apply self-edits using regular expressions and NLP
    for line in text.splitlines():
        if re.search(r'\bafraid\b', line, re.IGNORECASE):  # find patterns like "I am afraid"
            sentiment = sia.polarity_scores(line)
            if sentiment['compound'] < 0:  # negative sentiment
                edited_line = f"Remember, {line} is not a reflection of your true identity. You are loved and valued by God."
            else:
                edited_line = line.replace("afraid", "fearless")
        elif re.search(r'\bself-doubt\b', line, re.IGNORECASE):  # find self-doubt phrases
            edited_line = line.replace("self-doubt", "confidence in God's plan for you")
        else:
            edited_line = line

        edited_lines.append(edited_line)

    return '\n'.join(edited_lines)
