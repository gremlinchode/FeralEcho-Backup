import os
import stat

def check_permissions(file_path):
    file_stats = os.stat(file_path)
    permission_bits = file_stats.st_mode & 0o777

    if permission_bits & stat.S_IRUSR:
        print("Read permission granted!")
    elif permission_bits & stat.S_IWUSR:
        print("Write permission granted!")
    else:
        print("Permission denied!")

    return permission_bits
