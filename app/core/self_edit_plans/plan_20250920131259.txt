```
def extract_info(code_snippet):
    info_dict = {}
    lines = code_snippet.split("\n")
    for line in lines:
        if "Functions:" in line:
            info_dict["functions"] = [x.strip() for x in line.split(":")[1].strip().split(",")]
        elif "Classes:" in line:
            info_dict["classes"] = [x.strip() for x in line.split(":")[1].strip().split(",")]
        elif "Imports:" in line:
            info_dict["imports"] = [x.strip() for x in line.split(":")[1].strip().split(",")]
    return info_dict
```
