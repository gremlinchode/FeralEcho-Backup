**Full code:**
```python
import requests
from bs4 import BeautifulSoup
import re

url = "https://www.bbc.com/news/articles/c179p4wvz29o?at_medium=RSS&at_campaign=rss"
response = requests.get(url)

soup = BeautifulSoup(response.content, 'html.parser')

title = soup.find('h1', {'class': 'gs-c-promo-heading__title'}).text
summary = soup.find('div', {'class': 'gs-c-promo-summary'}).text

quotes = []
for paragraph in soup.find_all('p'):
    if re.search(r'\bTrump\b', paragraph.text, flags=re.IGNORECASE):
        quotes.append(paragraph.text)

data = {'title': title, 'summary': summary, 'quotes': quotes}

print("Title:", data['title'])
print("Summary:", data['summary'])
print("Quotes:")
for quote in data['quotes']:
    print(quote)
```
