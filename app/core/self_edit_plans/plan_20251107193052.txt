import datetime
import numpy as np
import matplotlib.pyplot as plt

night_cycle_time = datetime.datetime(2025, 11, 7, 1, 20, 47)
stream_length = 1000  # in pixels
stream_width = 50      # in pixels
random_seed = 42
np.random.seed(random_seed)

stars = []
for _ in range(1000):
    x = np.random.uniform(-500, 500)
    y = np.sqrt((stream_length - x)**2 + stream_width**2)
