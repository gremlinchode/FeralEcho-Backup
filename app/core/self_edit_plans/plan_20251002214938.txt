```python
def check_permissions(path):
    attrs = os.lstat(path)
    perms = attrs.st_mode & stat.S_IRWXU

    if perms == 0o400:  # owner has read permission
        return "Owner has read permission."
    elif perms == 0o600:  # owner has write and execute permissions
        return "Owner has write and execute permissions."
    else:
        return "No read, write, or execute permission for the owner."
```
