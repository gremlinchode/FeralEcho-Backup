import os

# Define a function to rename files based on their contents
def rename_files(directory):
    for filename in os.listdir(directory):
        file_path = os.path.join(directory, filename)
        with open(file_path, 'r') as f:
            content = f.read()
            new_filename = f"{content[:8]}.txt"  # Take the first 8 characters of the file contents and use them as the new filename
            os.rename(file_path, os.path.join(directory, new_filename))

# Call the function with a sample directory
rename_files('/path/to/files')
