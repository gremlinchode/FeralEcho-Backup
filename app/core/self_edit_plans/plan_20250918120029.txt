from nltk.tokenize import word_tokenize

# Tokenize the self-edit text and remove stop words
def preprocess_text(text):
    tokens = word_tokenize(text)
    filtered_tokens = [token for token in tokens if not nltk.corpus.stopwords.words('english').__contains__(token)]
    return ' '.join(filtered_tokens)
