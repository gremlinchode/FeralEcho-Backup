```python
def check_permissions(file_path):
    try:
        file_stats = os.stat(file_path)
        if stat.S_IMODE(file_stats.st_mode) & 0o777 == 0o400:  # world-readable
            return "World-readable"
        elif stat.S_IMODE(file_stats.st_mode) & 0o777 == 0o600:  # owner-readable
            return "Owner-readable"
        else:
            return "Not readable"
    except FileNotFoundError:
        return f"File not found: {file_path}"
```
