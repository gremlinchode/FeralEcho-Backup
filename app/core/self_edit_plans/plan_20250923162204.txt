```python
import requests
import json
from bs4 import BeautifulSoup

url = "https://en.wikipedia.org/w/api.php"
params = {"action": "query", "format": "json", "titles": "Artificial intelligence"}
response = requests.get(url, params=params)
data = json.loads(response.text)

page_id = data["query"]["pages"][1164]["id"]
revision_id = data["query"]["pages"][1164]["revisions"][1311410574]["id"]

html_response = requests.get("https://en.wikipedia.org/wiki/Artificial_intelligence")
soup = BeautifulSoup(html_response.content, 'html.parser')

title_tag = soup.find("h1", {"class": "mw-page-title-main"})
print(title_tag.text)

print(f"The title of the 'Artificial intelligence' page is: {title_tag.text}")
