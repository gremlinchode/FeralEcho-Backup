**Putting it all together**
Here's the complete code:

```python
import requests
from bs4 import BeautifulSoup

url = "https://www.bbc.com/news/articles/c179p4wvz29o?at_medium=RSS&at_campaign=rss"
response = requests.get(url)
article_html = response.content

soup = BeautifulSoup(article_html, 'html.parser')

title = soup.find('h1', class_='gs-c-promo-heading__title').text
article_text = ''
for paragraph in soup.find_all('p'):
    article_text += paragraph.text + '\n'

quotes = []
for blockquote in soup.find_all('blockquote'):
    quote_text = blockquote.text.strip()
    quotes.append(quote_text)

article_text = article_text.lower().replace('\n', ' ')

print(f"Title: {title}")
print(f"Article Text:\n{article_text}")
```
