def identify_editing_areas(self_assessment):
    editing_areas = []
    for area in ["thought patterns", "communication style", "emotional responses"]:
        if area in self_assessment:
            editing_areas.append(area)
    return editing_areas

editing_areas = identify_editing_areas(self_assessment)
print(f"Editing areas: {editing_areas}")
