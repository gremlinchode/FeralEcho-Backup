```python
def analyze_code(parsed_code):
    # Initialize an empty dictionary to store results
    analysis_results = {}
    
    # Iterate through each node in the parsed code tree
    for node in parsed_code:
        if isinstance(node, ast.FunctionDef):
            # If it's a function definition, extract its name and parameters
            func_name = node.name
            params = [arg.arg for arg in node.args.args]
            
            # Store the results in our analysis dictionary
            analysis_results[func_name] = {'params': params}
        elif isinstance(node, ast.ClassDef):
            # If it's a class definition, extract its name and attributes
            class_name = node.name
            attrs = [attr.arg for attr in node.bases]
            
            # Store the results in our analysis dictionary
            analysis_results[class_name] = {'attrs': attrs}
        else:
            # For all other nodes (e.g., imports), just store their type
            analysis_results[node.__class__.__name__] = None
    
    return analysis_results
```
