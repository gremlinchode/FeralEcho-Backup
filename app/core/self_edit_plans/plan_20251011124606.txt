```python
def check_permissions(file_path):
    try:
        perms = os.stat(file_path).st_mode
        if stat.S_IRWXU & perms:  # owner has read/write/execute permissions
            return "Owner has permissions"
        elif stat.S_IRWXG & perms:  # group has read/write/execute permissions
            return "Group has permissions"
        else:
            return "No permissions found"
    except FileNotFoundError:
        return f"{file_path} not found"
```
