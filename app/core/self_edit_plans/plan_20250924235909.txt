```python
def parse_code_analysis(code_analysis):
    # Initialize an empty dictionary to store the parsed code analysis
    parsed_data = {"Functions": [], "Classes": [], "Imports": []}

    # Iterate through the code analysis and populate the dictionary
    for line in code_analysis.splitlines():
        if line.startswith("Functions:"):
            parsed_data["Functions"] = [func.strip() for func in line[10:].split(",")]
        elif line.startswith("Classes:"):
            parsed_data["Classes"] = [cls.strip() for cls in line[8:].split(",")]
        elif line.startswith("Imports:"):
            parsed_data["Imports"] = [imp.strip() for imp in line[8:].split(",")]

    return parsed_data

def extract_keywords(parsed_data):
    keywords = {}
    for key, value in parsed_data.items():
        if value:
            keywords[key] = value
    return keywords
```

**Step 5: Test and Refine (Iterative Improvement)**
