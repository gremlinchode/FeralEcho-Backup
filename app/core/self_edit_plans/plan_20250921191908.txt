Here's the complete code:
```python
import json
import requests

api_endpoint = "https://api.stackexchange.com/2.2/questions"
params = {"order": "desc", "sort": "activity", "site": "stackoverflow"}

response = requests.get(api_endpoint, params=params)
data = json.loads(response.content)

questions = []
for item in data['items']:
    questions.append({
        'question_id': item['question_id'],
        'title': item['title']
    })

trakted_questions = [q for q in questions if 'pyside6' in q['title'].lower()]

print(trakted_questions)
```
