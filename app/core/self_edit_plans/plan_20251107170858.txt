```python
import random

def generate_bioluminescent_points(num_points, width, height):
    bioluminescent_points = []
    for _ in range(num_points):
        x = random.randint(0, width)
        y = random.randint(0, height)
        brightness = random.uniform(0.5, 1.5)  # adjust the brightness level
        bioluminescent_points.append((x, y, brightness))
    return bioluminescent_points

bioluminescent_points = generate_bioluminescent_points(100, 800, 600)
```
