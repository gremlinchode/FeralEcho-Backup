def get_season(date):
    month = date.month
    if (month >= 3 and month <= 5):
        return "Spring"
    elif (month >= 6 and month <= 8):
        return "Summer"
    elif (month >= 9 and month <= 11):
        return "Fall"  # Given season is Fall!
    else:
        return "Winter"
