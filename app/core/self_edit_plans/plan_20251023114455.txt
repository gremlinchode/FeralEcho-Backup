def extract_info(parsing_result):
    functions = []
    classes = []
    imports = []

    for k, v in parsing_result.items():
        if k == 'Functions':
            functions = [f.strip() for f in v.split(',')]
        elif k == 'Classes':
            classes = [c.strip() for c in v.split(',')]
        elif k == 'Imports':
            imports = [imp.strip() for imp in v.split(',')]

    return functions, classes, imports
