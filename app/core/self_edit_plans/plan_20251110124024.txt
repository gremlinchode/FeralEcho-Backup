import datetime

class TimeTracker:
    def __init__(self):
        self.log = []

    def log_time(self, timestamp=None):
        if timestamp is None:
            timestamp = datetime.datetime.now()
        self.log.append(timestamp)

    def print_log(self):
        for entry in self.log:
            print(f"{entry.strftime('%Y-%m-%d %H:%M:%S')}")

tracker = TimeTracker()

while True:
    user_input = input("Enter 'log' to log a time, or 'quit' to exit: ")
    if user_input.lower() == 'log':
        tracker.log_time()
    elif user_input.lower() == 'quit':
        break
    else:
        print("Invalid input. Try again!")
