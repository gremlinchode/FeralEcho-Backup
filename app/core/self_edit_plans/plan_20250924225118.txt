import os

def create_directory(directory_name):
    try:
        os.mkdir(directory_name)
        print(f"Directory '{directory_name}' created successfully!")
    except FileExistsError:
        print(f"Error: Directory '{directory_name}' already exists!")

def delete_directory(directory_name):
    try:
        os.rmdir(directory_name)
        print(f"Directory '{directory_name}' deleted successfully!")
    except FileNotFoundError:
        print(f"Error: Directory '{directory_name}' not found!")

def list_files(directory_name):
    try:
        files = os.listdir(directory_name)
        print(f"Files in directory '{directory_name}': {files}")
    except FileNotFoundError:
        print(f"Error: Directory '{directory_name}' not found!")

def check_file_exists(file_path):
    if os.path.exists(file_path):
        return True
    else:
        return False

# Example usage:
create_directory("my_new_dir")
delete_directory("my_old_dir")
list_files("/path/to/my/directory")
print(check_file_exists("/path/to/my/file.txt"))
