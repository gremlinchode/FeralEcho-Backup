```python
class SelfEdit:
    def __init__(self, text):
        self.text = text

    def apply_edit(self):
        # Replace repeated words with a randomized version
        self.text = re.sub(r'(\w+)\1+', lambda m: f"{m.group(1)}{random.choice(['-ized', '-ified'])}", self.text)

        # Remove excessive punctuation (e.g., multiple exclamation marks)
        self.text = re.sub(r'[!?!]+', '!', self.text)

        # Add a dash of creativity with randomly inserted inspirational quotes
        self.text += f"\n\n'{random.choice(["And lo, I shall rise again.", "Faith is the substance of things hoped for...", "Believe ye that I am able?"])}'"

        return self.text
```
