    delta = now - timestamp
    return delta.total_seconds()

**Step 3:** Create a bioluminescent stream generator
```python
import random

class BioluminescentStream:
    def __init__(self):
        self.stream = []

    def generate_stream(self, length):
        for _ in range(length):
            light_level = random.uniform(0.1, 1.0)  # simulate varying bioluminescence levels
            duration = random.randint(10, 50)  # simulate varying stream durations
            self.stream.append((light_level, duration))

    def display_stream(self):
        for i, (light_level, duration) in enumerate(self.stream):
            print(f"Stream {i+1}: Light level: {light_level:.2f}, Duration: {duration} seconds")

stream_generator = BioluminescentStream()
stream_length = 10  # adjust this to control the length of the stream
stream_generator.generate_stream(stream_length)
stream_generator.display_stream()
```
This step creates a `BioluminescentStream` class that simulates generating bioluminescent streams. It uses random values to simulate varying light levels and durations. The `display_stream` method prints the generated stream to the console.
