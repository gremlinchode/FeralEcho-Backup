def refine_clarity(text):
    clarity_issues = []
    for sentence in text.split("."):
        if len(sentence) > 20:  # arbitrary threshold
            clarity_issues.append((sentence, "Too long"))
        elif any(word not in ["the", "and", "a"] for word in sentence.split()):
            clarity_issues.append((sentence, "Unclear"))
    return [issue[0] + " (suggested edit: " + issue[1] + ")" for issue in clarity_issues]
```

import textblob

def adjust_tone(text):
    tone_issues = []
    blob = textblob.TextBlob(text)
    for sentence in text.split("."):
        if blob.sentiment.polarity < 0:  # negative sentiment
            tone_issues.append((sentence, "Too negative"))
        elif any(word.lower() in ["formal", "stuffy"] for word in sentence.split()):
            tone_issues.append((sentence, "Too formal"))
    return [issue[0] + " (suggested edit: " + issue[1] + ")" for issue in tone_issues]
```
