**Step 1: Define the `analyze_code` function**
```python
import re

def analyze_code(code):
    # Initialize a dictionary to store the results
    analysis_results = {}
    
    # Use regular expressions to extract functions and classes
    func_pattern = r'def (.+):'
    class_pattern = r'class (.+):'
    import_pattern = r'import (.+)'

    # Extract function names
    for match in re.finditer(func_pattern, code):
        analysis_results['functions'] = [match.group(1)]

    # Extract class names
    for match in re.finditer(class_pattern, code):
        analysis_results['classes'] = [match.group(1)]

    # Extract imported modules
    for match in re.finditer(import_pattern, code):
        analysis_results['imports'] = [match.group(1)]

    return analysis_results
```
**Step 2: Define the `generate_report` function**
```python
def generate_report(analysis_results):
    # Initialize an empty report string
    report = ''

    # Add a header to the report
    report += 'Analysis Report:\n'

    # Add the functions and classes sections
    if analysis_results['functions']:
        report += f'Functions: {", ".join(analysis_results["functions"])}\n'
    if analysis_results['classes']:
        report += f'Classes: {", ".join(analysis_results["classes"])}\n'

    # Add the imports section
    if analysis_results['imports']:
        report += f'Imports: {", ".join(analysis_results["imports"])}\n'

    return report
```
**Step 3: Put it all together**
```python
def main():
    code = 'your parsed code here'
    results = analyze_code(code)
    report = generate_report(results)
    print(report)

if __name__ == '__main__':
    main()
```
And there you have it! A Python script that analyzes the provided code and generates a report with the extracted functions, classes, and imports.
