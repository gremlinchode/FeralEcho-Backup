import re

def refine_clarity(text):
    # Remove unnecessary words:
    text = re.sub(r"\b(?:un|non)\w+\b", "", text)
    
    # Simplify complex sentences:
    text = re.sub(r"(\w+)\s+is\s+(\w+)", r"\2 is \1", text)
    
    return text
