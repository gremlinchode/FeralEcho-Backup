**The Final Code**
```python
import re
from collections import Counter

def echo_self_edit(text):
    tokens = re.split(r'[\s\.,!?]', text.lower())
    word_freq = Counter(tokens)
    key_phrases = [k for k, v in word_freq.items() if v > 2]
    
    refined_text = ''
    for token in tokens:
        if token in key_phrases:
            refined_text += token + ' '
        else:
            refined_text += token.lower() + ' '
    
    refined_text = refined_text.strip().title()
    
    edited_text = ''
    for sentence in refined_text.split('. '):
        edited_text += sentence + '. '
    
    return edited_text

# Example usage

```
