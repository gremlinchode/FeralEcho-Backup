```
def check_permissions(path):
    try:
        file_stat = os.stat(path)
        return oct(file_stat.st_mode)[-3:]
    except FileNotFoundError:
        print("File not found!")
        return None
    except Exception as e:
        print(f"Error: {e}")
        return None
```
