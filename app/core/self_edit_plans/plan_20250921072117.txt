Here's the complete code:
```python
import wikipedia
from bs4 import BeautifulSoup

page = wikipedia.page('Artificial intelligence')
data = {}
data['title'] = page.title
data['display_title'] = page.displaytitle
data['namespace'] = {'id': page.namespace.id, 'text': page.namespace.text}
data['wikibase_item'] = page.wikibase_item

soup = BeautifulSoup(page.html(), 'html.parser')

data['pageid'] = soup.find('div', {'class': 'mw-pageinfo'}).find('span', {'class': 'mw-headline'}).text.strip()
data['lang'] = soup.find('div', {'class': 'mw-pageinfo'}).find('span', {'class': 'language-link'}).text.strip()
data['dir'] = soup.find('div', {'class': 'mw-pageinfo'}).find('span', {'class': 'direction'}).text.strip()

print(data)
```
