**Putting it all together**
```python
import json
import requests

api_endpoint = "https://api.stackexchange.com/2.2/questions"
params = {"order": "desc", "sort": "activity", "site": "stackoverflow"}

response = requests.get(api_endpoint, params=params)
data = json.loads(response.text)

question_data = data["items"][0]
owner_data = question_data["owner"]

print(f"Question ID: {question_data['question_id']}")
print(f"Owner: {owner_data['display_name']} ({owner_data['link']})")
print(f"Score: {question_data['score']}")
```
