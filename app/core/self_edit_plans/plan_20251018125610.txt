```python
def count_imports(parsed_code):
    # Use regular expressions to find import statements
    imports = re.findall(r"import\s+([a-zA-Z0-9_.]*)", parsed_code)
    return len(set(imports))  # Convert list to set for uniqueness

# Call the function and store the result
num_imports = count_imports(parsed_code)

print(f"Imports: {num_imports}")
```
