```python
def game_loop():
    # Initialize our game variables
    score = 0
    lives = 5
    
    while lives > 0:
        # Present the current game state to the player
        print("Score:", score)
        print("Lives:", lives)
        
        # Ask the player for input (e.g., a move or action)
        user_input = input("> ")
        
        # Process the user's input and update the game state accordingly
        if user_input == "move_left":
            score += 1
        elif user_input == "jump":
            lives -= 1
        else:
            print("Invalid input. Try again!")
        
        # Check for game over (e.g., when lives reach zero)
        if lives <= 0:
            print("Game Over! Your final score was:", score)
            break
    
    # End the game loop and display a congratulatory message
    if lives > 0:
        print(" Congratulations, you won with a score of", score)

game_loop()
```
