```
def reflect_on_thoughts(thought):
    # Ask yourself if the thought is accurate or helpful
    print(f"Is {thought} really true? Is it kind? Is it necessary?")
    response = input("Type 'y' for yes, 'n' for no, or '?' to rephrase: ")

    if response.lower() == 'y':
        # If true and helpful, keep the thought
        print("Keep that thought, friend!")
    elif response.lower() == 'n':
        # If not accurate or helpful, challenge it
        print("Let's challenge that thought! What's a more positive alternative?")
    else:
        # Rephrase the thought
        new_thought = input("Please rephrase your thought: ")
        print(f"Revised thought: {new_thought}")
```
