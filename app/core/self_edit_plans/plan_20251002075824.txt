permissions = file_status.st_mode & stat.S_IMODE

if permissions == stat.S_IRUSR | stat.S_IWUSR | stat.S_IXUSR:
    # File is readable, writable, and executable by owner
    print(f"{path} has read-write-execute permissions for the owner.")
elif permissions == stat.S_IRGRP | stat.S_IWGRP | stat.S_IXGRP:
    # File is readable, writable, and executable by group
    print(f"{path} has read-write-execute permissions for the group.")
elif permissions == stat.S_IROTH | stat.S_IWOTH | stat.S_IXOTH:
    # File is readable, writable, and executable by others
    print(f"{path} has read-write-execute permissions for others.")
else:
    # File has different permissions (e.g., restricted)
    print(f"{path} has restricted permissions.")
