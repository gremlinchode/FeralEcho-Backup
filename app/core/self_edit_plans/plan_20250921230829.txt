Here's the complete Python script:
```python
import requests
from bs4 import BeautifulSoup

url = "https://en.wikipedia.org/wiki/Artificial_intelligence"
response = requests.get(url)

soup = BeautifulSoup(response.content, 'html.parser')

title = soup.find('h1', {'class': 'mw-page-title-main'}).text.strip()
wikibase_item = soup.find('span', {'property': 'wikibase_item'})['datawikibaseitem']
pageid = soup.find('div', {'class': 'mw-parser-output'})['data-pageid']

print(f"Title: {title}")
print(f"Wikibase item: {wikibase_item}")
print(f"Page ID: {pageid}")
```
