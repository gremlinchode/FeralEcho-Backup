def get_season(year, month):
    if (month >= 3 and month <= 5) or (month == 12 and year % 4 == 0):
        return "Spring"
    elif (month >= 6 and month <= 8):
        return "Summer"
    elif (month >= 9 and month <= 11):
        return "Autumn"  # Ah, Fall!
    else:
        return None

season = get_season(2025, 11)
