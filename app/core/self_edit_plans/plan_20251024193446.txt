import random

# Define a function to simulate bioluminescent stream patterns
def simulate_streams(num_points, duration):
    points = []
    for _ in range(num_points):
        x = random.uniform(0, 10)
        y = random.uniform(0, 5)
        z = random.uniform(0, 2)
        points.append((x, y, z))

    # Calculate the time elapsed and the duration of each point
    time_elapsed = [i / (num_points * duration) for i in range(num_points)]
    durations = [(time_elapsed[i], points[i]) for i in range(num_points)]

    return durations

# Simulate 10 bioluminescent streams for 5 seconds
streams = simulate_streams(10, 5)
