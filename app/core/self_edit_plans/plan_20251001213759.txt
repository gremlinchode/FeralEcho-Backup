```python
def check_permissions(file_path: str, required_permission: int) -> bool:
    try:
        stat_info = os.stat(file_path)
        return stat.S_IMODE(stat_info.st_mode) & required_permission == required_permission
    except OSError as e:
        print(f"Error checking permissions for {file_path}: {e}")
        return False
```
