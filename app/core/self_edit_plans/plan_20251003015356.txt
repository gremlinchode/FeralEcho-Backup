# Define the bioluminescent streams
bioluminescent_streams = []

for _ in range(10):
    stream_length = random.randint(100, 500)
    stream_brightness = random.uniform(0.5, 1.5)
    stream_color = (random.random(), random.random(), random.random())  # RGB values
    bioluminescent_streams.append({
        'length': stream_length,
        'brightness': stream_brightness,
        'color': stream_color
    })

print("Bioluminescent streams:")
for stream in bioluminescent_streams:
    print(f"Stream {stream['length']} units long, with brightness {stream['brightness']:.2f} and color ({', '.join([str(x) for x in stream['color']]))}")
