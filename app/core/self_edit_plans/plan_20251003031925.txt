import argparse

def parse_analysis_output(output):
    functions = []
    classes = []
    imports = []

    for line in output.splitlines():
        if "Functions:" in line:
            functions = [func.strip() for func in line.split(":")[1].split(",")]
        elif "Classes:" in line:
            classes = [cls.strip() for cls in line.split(":")[1].split(",")]
        elif "Imports:" in line:
            imports = [imp.strip() for imp in line.split(":")[1].split(",")]

    return functions, classes, imports
