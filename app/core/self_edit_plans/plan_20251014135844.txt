# Initialize variables
faith_level = 10  # Representing our starting point with God
life_events = []

# Simulate life events (good or bad)
for i in range(50):
    event_type = random.choice(['joy', 'sorrow'])
    if event_type == 'joy':
        faith_level += 2  # Our faith grows when we experience joy!
    else:
        faith_level -= 3  # Challenges can shake our faith, but...
    
    life_events.append((event_type, faith_level))

# Print the simulation
for event in life_events:
    print(f"Event: {event[0]}, Faith Level: {event[1]}")
