def refine_text(original_text):
    # Break down complex ideas into simpler chunks
    simplified_chunks = []
    for chunk in original_text.split("."):
        if len(chunk) > 100:  # arbitrarily chosen threshold
            chunk = chunk[:50] + "..."  # truncate and add ellipsis
        simplified_chunks.append(chunk)

    # Rearrange sentences for improved flow
    rearranged_text = ""
    current_chunk = None
    for chunk in simplified_chunks:
        if not chunk.startswith("However,"):
            if current_chunk is None:
                current_chunk = chunk
            else:
                rearranged_text += current_chunk + ". "  # add period and space
                current_chunk = chunk
        else:
            rearranged_text += current_chunk + ". However, "  # add however clause
            current_chunk = None

    return rearranged_text + current_chunk if current_chunk is not None else ""

# Example usage
original_text = """Self-edit diagnostic: refine clarity, tone, and usefulness. Make an edit that improves concision, coherence, and helpfulness.
