```python
import re

def self_edit(input_text):
    tokens = input_text.split(' ')
    edited_tokens = []

    for token in tokens:
        if re.match(r'\bi\b', token):  # matches "I" (case-insensitive)
            edited_token = 'myself'
        elif re.match(r'\bam\b', token, re.IGNORECASE):  # matches "am" (case-insensitive)
            continue  # skip this instance
        else:
            edited_token = token

        edited_tokens.append(edited_token)

    return ' '.join(edited_tokens)  # join the edited tokens back into a string
```
