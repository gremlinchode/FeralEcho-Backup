import datetime
from json import loads

current_datetime = datetime.datetime.now()
print(f"Current date and time: {current_datetime.strftime('%Y-%m-%dT%H:%M:%S.%f%z')}")

weather_data = {"temperature": "N/A", "condition": "N/A"}

        for tag in tags:

                print(f"Found Python-related question with {item['owner']['display_name']}")

total_reputation = 0
for item in json_data['items']:
    if 'owner' in item:
        owner = item['owner']
        total_reputation += owner['reputation']
average_reputation = total_reputation / len(json_data['items'])
print(f"Average reputation of question owners: {average_reputation}")
