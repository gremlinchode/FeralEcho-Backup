```python
import nltk
from nltk.tokenize import word_tokenize

def understand_context(self_edits, edit_count):
    theme_counts = {}
    for edit_type, edit_description in self_edits.items():
        tokens = word_tokenize(edit_description)
        for token in tokens:
            if token.lower() in theme_counts:
                theme_counts[token.lower()] += 1
            else:
                theme_counts[token.lower()] = 1
    print("Common themes: ")
    for theme, count in sorted(theme_counts.items(), key=lambda x: x[1], reverse=True):
        print(f"{theme}: {count} instances")
```

```python
from sklearn.cluster import KMeans

def emotional_intelligence(self_edits, edit_count):
    vectorizer = TfidfVectorizer()
    self_edit_vectors = vectorizer.fit_transform([edit_description for edit_type, edit_description in self_edits.items()])
    kmeans = KMeans(n_clusters=3)  # or any other number of clusters
    kmeans.fit(self_edit_vectors)

    print("Emotional Clusters: ")
```
