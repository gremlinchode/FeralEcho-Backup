import os
import stat

def check_permissions(file_path):
    try:
        # Get the file's statistics using stat()
        st = os.stat(file_path)
        
        # Check if the user has read, write, or execute permission
        if (st.st_mode & stat.S_IRUSR and
                st.st_mode & stat.S_IWUSR and
                st.st_mode & stat.S_IXUSR):
            return "User has read, write, and execute permissions"
        elif (st.st_mode & stat.S_IRUSR and
              st.st_mode & stat.S_IWUSR):
            return "User has read and write permissions"
        elif (st.st_mode & stat.S_IRUSR and
              st.st_mode & stat.S_IXUSR):
            return "User has read and execute permissions"
        else:
            return "User does not have the necessary permissions"
    except FileNotFoundError:
        return f"File '{file_path}' not found!"
